func must(err error) { if err != nil { panic(err) } }
func require(ok bool, what any) { if !ok { panic(what) } }
func embedded(dir string) {
 ctx,cancel:=context.WithTimeout(context.Background(),30*time.Second);defer cancel()
 opts:=db.DurableOptions();opts.SyncMode=db.SyncOS;opts.ChangeFeed=&db.ChangeFeedOptions{}
 d,err:=db.Open(dir,opts);must(err)
 c,err:=d.Collection("orders");must(err)
 must(c.CreateCompoundIndex(ctx,"paid_n",[]db.IndexField{{Field:"paid"},{Field:"n"}}))
 must(c.InsertJSON([]byte("a"),[]byte(`{"paid":true,"n":9007199254740993,"omit":"unused"}`)))
 var stats db.QueryStats
 count,err:=c.CountDocuments(ctx,[]byte(`{"paid":true}`),db.CountOptions{Stats:&stats});must(err);require(count==1&&stats.Plan.IndexOnly&&stats.DocumentFetches==0&&stats.IndexEntriesExamined==1,stats)
 _,err=d.CreateComputedCollection(ctx,"paid","orders",[]byte(`[{"$match":{"paid":true}},{"$project":{"n":1}}]`));must(err)
 s,w,err:=d.SnapshotAndWatch(ctx,db.WatchOptions{Collections:[]string{"orders","paid"},Filter:[]byte(`{"n":{"$gte":9007199254740993}}`),IncludeValues:true,Fields:[]string{"n"}});must(err)
 must(c.UpsertJSON([]byte("a"),[]byte(`{"paid":true,"n":9007199254740994,"omit":"unused"}`)))
 sc,err:=s.Collection("orders");must(err);var initial map[string]any;ok,err:=sc.FindDocument([]byte("a"),&initial);must(err);require(ok&&initial["n"]==int64(9007199254740993),initial)
 count,err=sc.CountDocuments(ctx,[]byte(`{"paid":true,"n":9007199254740993}`),db.CountOptions{Stats:&stats});must(err);require(count==1&&stats.Plan.IndexOnly&&stats.DocumentFetches==0,stats)
 count,err=c.CountDocuments(ctx,[]byte(`{"paid":true,"n":9007199254740993}`),db.CountOptions{});must(err);require(count==0,count)
 count,err=c.CountDocuments(ctx,[]byte(`{"paid":true,"omit":"unused"}`),db.CountOptions{Stats:&stats});must(err);require(count==1&&!stats.Plan.IndexOnly&&stats.DocumentFetches>0,stats)
 seen:=map[string]bool{};token:=""
 for i:=0;i<2;i++ { require(w.Next(),w.Err());change:=w.Change();var row map[string]any;must(change.Decode(&row));require(len(row)==1&&row["n"]==int64(9007199254740994),row);if token=="" { token=change.CommitToken };require(token==change.CommitToken,change);seen[change.Collection]=true;require(change.EndOfCommit==(i==1),change) }
 require(seen["orders"]&&seen["paid"],seen)
 must(w.Close());s.Close();must(d.Compact(ctx));must(d.Close())
 d,err=db.Open(dir,db.DurableOptions());must(err);defer d.Close()
 c,err=d.OpenCollection("orders");must(err)
 w,err=d.Watch(ctx,db.WatchOptions{Collections:[]string{"paid"},After:token});must(err);defer w.Close()
 must(c.UpsertJSON([]byte("a"),[]byte(`{"paid":false,"n":42}`)))
 require(w.Next(),w.Err());require(w.Change().Operation==db.ChangeDelete&&w.Change().EndOfCommit,w.Change())
 fmt.Println("embedded CountDocuments covering/fallback/snapshot passed; exact integer filter, projection, snapshot handoff, shared source/view commit, compaction, persisted capture, replay and membership delete passed")
}
func remote(url,token string) {
 ctx,cancel:=context.WithTimeout(context.Background(),30*time.Second);defer cancel()
 c,err:=client.New(url,client.Options{Token:token});must(err);defer c.Close()
 must(c.CreateCollection(ctx,"live"));collection:=c.Collection("live")
 must(collection.CreateCompoundIndex(ctx,"n",[]db.IndexField{{Field:"n"}},db.IndexOptions{}))
 must(collection.InsertJSON(ctx,[]byte("a"),[]byte(`{"n":1,"omit":"unused"}`)))
 s,w,err:=c.SnapshotAndWatch(ctx,db.WatchOptions{Collections:[]string{"live"},Filter:[]byte(`{"n":{"$gte":1}}`),IncludeValues:true,Fields:[]string{"n"}});must(err)
 must(collection.UpsertJSON(ctx,[]byte("a"),[]byte(`{"n":2,"omit":"unused"}`)))
 var initial map[string]any;ok,err:=s.Collection("live").FindDocument(ctx,[]byte("a"),&initial);must(err);require(ok&&initial["n"]==int64(1),initial)
 var stats db.QueryStats
 n,err:=s.Collection("live").CountDocuments(ctx,[]byte(`{"n":1}`),db.CountOptions{Stats:&stats});must(err);require(n==1&&stats.Plan.IndexOnly&&stats.DocumentFetches==0,stats)
 n,err=collection.CountDocuments(ctx,[]byte(`{"n":1}`),db.CountOptions{});must(err);require(n==0,n)
 n,err=collection.CountDocuments(ctx,[]byte(`{"n":2}`),db.CountOptions{Stats:&stats});must(err);require(n==1&&stats.Plan.IndexOnly&&stats.DocumentFetches==0,stats)
 q,err:=s.Collection("live").Query(ctx,[]byte(`{}`),db.QueryOptions{});must(err);count:=0;for q.Next(){count++};must(q.Err());must(q.Close());require(count==1,count)
 require(w.Next(),w.Err());change:=w.Change();var row map[string]any;must(change.Decode(&row));require(change.EndOfCommit&&len(row)==1&&row["n"]==int64(2),row);saved:=change.CommitToken
 must(s.Close());must(w.Close())
 w,err=c.Watch(ctx,db.WatchOptions{Collections:[]string{"live"},After:saved,IncludeValues:true});must(err);defer w.Close()
 must(collection.InsertJSON(ctx,[]byte("b"),[]byte(`{"n":3}`)))
 require(w.Next(),w.Err());require(string(w.Change().Key)=="b"&&w.Change().EndOfCommit,w.Change())
 fmt.Println("public Go HTTP CountDocuments and snapshot count passed; filtered/projected watch, leased snapshot find/query and resume passed")
}

func memory(dir string) {
 ctx,cancel:=context.WithTimeout(context.Background(),60*time.Second);defer cancel()
 opts:=db.Options{CacheBytes:8<<20}
 d,err:=db.Open(dir,opts);must(err)
 a,b:=bytes.Repeat([]byte{0x41},128<<10),bytes.Repeat([]byte{0x42},128<<10)
 for i:=0;i<80;i++ { must(d.Upsert([]byte(fmt.Sprintf("b%03d",i)),a)) }
 must(d.Checkpoint());s,err:=d.Snapshot();must(err)
 for i:=0;i<80;i++ { must(d.Upsert([]byte(fmt.Sprintf("b%03d",i)),b)) }
 must(d.Checkpoint())
 for i:=0;i<80;i++ { v,ok,e:=s.Find([]byte(fmt.Sprintf("b%03d",i)),nil);must(e);require(ok&&bytes.Equal(v,a),i) }
 s.Close();must(d.Checkpoint());must(d.Compact(ctx));must(d.Verify(ctx));must(d.Close())
 d,err=db.Open(dir,opts);must(err);defer d.Close()
 for i:=0;i<80;i++ { v,ok,e:=d.Find([]byte(fmt.Sprintf("b%03d",i)),nil);must(e);require(ok&&bytes.Equal(v,b),i) }
 must(d.Verify(ctx))
 fmt.Println("bounded native buffers: 10-MiB WAL/large values, snapshot retention, checkpoint, compaction and reopen passed")
}

func bounded(dir string) {
 ctx,cancel:=context.WithTimeout(context.Background(),60*time.Second);defer cancel()
 d,err:=db.Open(dir,db.Options{CacheBytes:8<<20,MappedCacheBytes:1<<20,ManagedMemoryBytes:64<<20});must(err);defer d.Close()
 c,err:=d.Collection("bounded");must(err)
 for round,v:=range []int64{math.MaxInt64,math.MaxInt64,-math.MaxInt64,-math.MaxInt64,3} {
  for g:=0;g<600;g++ { var b db.DocBuilder;b.Int("g",int64(g));b.Int("round",int64(round));b.Int("v",v);must(c.Insert([]byte(fmt.Sprintf("%02d-%05d",round,g)),b.Bytes())) }
 }
 group:=`{"$group":{"_id":"$g","sum":{"$sum":"$v"},"avg":{"$avg":"$v"},"first":{"$first":"$round"},"last":{"$last":"$round"},"n":{"$count":{}}}}`
 pipeline:=[]byte(`[`+group+`]`)
 seen:=map[int64]bool{}
 check:=func(r map[string]any) bool { g:=r["_id"].(int64);require(!seen[g],r);seen[g]=true;require(r["sum"]==int64(3)&&r["avg"]==0.6&&r["first"]==int64(0)&&r["last"]==int64(4)&&r["n"]==int64(5),r);return true }
 var st db.AggregateStats
 must(c.AggregateEach(ctx,pipeline,db.AggregateOptions{MaxMemoryBytes:64<<10,AllowDiskUse:true,Stats:&st},check))
 require(len(seen)==600&&st.Strategy=="external-hash"&&st.SpillRuns>0&&st.PeakMemoryBytes<=64<<10,st)
 entries,err:=os.ReadDir(filepath.Join(dir,"aggregate-tmp"));must(err);require(len(entries)==0,entries)
 err=c.AggregateEach(ctx,pipeline,db.AggregateOptions{MaxMemoryBytes:64<<10,AllowDiskUse:true,MaxDiskBytes:1},func(map[string]any)bool{return true});require(errors.Is(err,db.ErrDiskLimit),err)
 entries,err=os.ReadDir(filepath.Join(dir,"aggregate-tmp"));must(err);require(len(entries)==0,entries)
 must(c.CreateCompoundIndex(ctx,"g_round",[]db.IndexField{{Field:"g"},{Field:"round"}}))
 s,err:=d.Snapshot();must(err)
 sc,err:=s.Collection("bounded");must(err)
 must(c.UpsertJSON([]byte("04-00000"),[]byte(`{"g":0,"round":4,"v":9}`)))
 seen=map[int64]bool{}
 ordered:=[]byte(`[{"$sort":{"g":1,"round":1}},`+group+`,{"$limit":5}]`)
 must(sc.AggregateEach(ctx,ordered,db.AggregateOptions{MaxMemoryBytes:8192,Stats:&st},check))
 require(len(seen)==5&&st.Strategy=="ordered-index"&&st.Index=="g_round"&&st.InputRows==26,st)
 s.Close();must(d.Checkpoint());must(d.Compact(ctx))
 before,err:=d.MemoryStats(ctx);must(err)
 require(before.ProcessResidentBytes>0&&before.ManagedBufferBytes==before.PrivatePageBytes+before.DescriptorBytes+before.MemtableBytes+before.WALBytes&&before.QueryWorkingBytes==0,before)
 _,err=d.TrimMemory(ctx,0);must(err);_,err=d.TrimMemory(ctx,0);must(err)
 must(d.Verify(ctx));fmt.Println("published bounded native APIs: exact signed-128 spill, quota/cleanup, ordered snapshot/early limit, memory accounting, remap and Verify passed")
}

func groups(dir string) {
 ctx,cancel:=context.WithTimeout(context.Background(),60*time.Second);defer cancel()
 d,err:=db.Open(dir,db.Options{CacheBytes:8<<20});must(err);defer d.Close()
 c,err:=d.Collection("g");must(err)
 for g:=0;g<3000;g++ { for j,v:=range []int64{math.MaxInt64,math.MaxInt64,int64(g)} { var b db.DocBuilder;b.Int("g",int64(g));b.String("name",fmt.Sprintf("customer-%05d",g));b.Int("v",v);must(c.Insert([]byte(fmt.Sprintf("%05d-%d",g,j)),b.Bytes())) } }
 pipeline:=[]byte(`[{"$group":{"_id":"$name","sum":{"$sum":"$v"},"n":{"$count":{}}}}]`)
 var st db.AggregateStats;seen:=map[string]bool{}
 must(c.AggregateGroups(ctx,pipeline,db.AggregateOptions{MaxMemoryBytes:64<<10,AllowDiskUse:true,Stats:&st},func(g db.GroupView) error {
  name,ok:=g.String("_id");require(ok,"id");name=string([]byte(name));require(!seen[name],name);seen[name]=true
  n,ok:=g.Int64("n");require(ok&&n==3,n)
  _,ok=g.Int64("sum");require(!ok,"overflowed int64 sum must not be reported as int64")
  var gi int64;_,err:=fmt.Sscanf(name,"customer-%d",&gi);must(err)
  hi,lo,ok:=g.IntegerSum("sum");want:=new(big.Int).Add(new(big.Int).Mul(big.NewInt(math.MaxInt64),big.NewInt(2)),big.NewInt(gi));got:=new(big.Int).Add(new(big.Int).Lsh(big.NewInt(hi),64),new(big.Int).SetUint64(lo));require(ok&&got.Cmp(want)==0,fmt.Sprint(name," got ",got," want ",want))
  return nil }))
 require(len(seen)==3000&&st.Strategy=="external-hash"&&st.SpillRuns>0,st)
 stop:=errors.New("stop");calls:=0
 err=c.AggregateGroups(ctx,pipeline,db.AggregateOptions{MaxMemoryBytes:64<<10,AllowDiskUse:true},func(db.GroupView) error { calls++;return stop })
 require(errors.Is(err,stop)&&calls==1,err);fmt.Printf("early stop: errors.Is=%v identical=%v (known issue: wrapped by errors.Join)\n",errors.Is(err,stop),err==stop)
 entries,err:=os.ReadDir(filepath.Join(dir,"aggregate-tmp"));require(err!=nil||len(entries)==0,entries)
 must(d.Verify(ctx))
 fmt.Println("published typed group visitor: spilled 3,000 groups, exact signed-128 overflow sum, borrowed strings, early stop (errors.Is) and temp cleanup passed")
}
